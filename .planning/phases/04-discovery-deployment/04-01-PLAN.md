   ```rust
   use pkarr::{Client, Keypair, SignedPacket};
   use anyhow::Result;

   pub struct PkarrPublisher {
       client: Client,
       pub keypair: Keypair,  // pub for main.rs to read public key for logging
   }

   impl PkarrPublisher {
       pub fn new(keypair: Keypair) -> Result<Self> {
           let client = Client::builder().build()?;
           Ok(Self { client, keypair })
       }

       pub async fn publish_record(&self, packet: SignedPacket) -> Result<()> {
           self.client.publish(&packet, None).await
               .map_err(|e| anyhow::anyhow!("PKARR publish failed: {e}"))
       }
   }

   /// Load existing keypair from file, or generate and persist a new one.
   /// The public key is the coordinator's stable DHT identity — never rotate it.
   pub fn load_or_generate_keypair(path: &str) -> Result<Keypair> {
       match Keypair::from_secret_key_file(path) {
           Ok(kp) => {
               tracing::info!("Loaded existing PKARR keypair from {path}");
               Ok(kp)
           }
           Err(_) => {
               let kp = Keypair::random();
               kp.write_secret_key_file(path)
                   .map_err(|e| anyhow::anyhow!("Cannot write PKARR key file at {path}: {e}"))?;
               tracing::info!(
                   pubkey = %kp.public_key().to_z32(),
                   "Generated new PKARR keypair, saved to {path}"
               );
               Ok(kp)
           }
       }
   }

   /// Build a signed DNS TXT packet containing coordinator metadata.
   ///
   /// Uses label "_blindjoin" with a single compact JSON value (fits in 255 bytes).
   /// TTL = 300s (matches heartbeat interval).
   pub fn build_coordinator_packet(
       keypair: &Keypair,
       coordinator_addr: &str,
       denomination_sats: u64,
       min_participants: u32,
       status: &str,
   ) -> Result<SignedPacket> {
       let record = serde_json::json!({
           "type": "blindjoin-coordinator",
           "version": "0.1.0",
           "onion": coordinator_addr,  // Phase 4: clearnet addr; Phase 5: .onion
           "network": "signet",
           "denomination_sats": denomination_sats,
           "min_participants": min_participants,
           "status": status,
       });
       let json_str = serde_json::to_string(&record)?;
       // DNS TXT string limit is 255 bytes. Log warning if close.
       if json_str.len() > 220 {
           tracing::warn!(len = json_str.len(), "PKARR TXT record approaching 255-byte DNS limit");
       }
       let ttl: u32 = 300;
       SignedPacket::builder()
           .txt("_blindjoin".try_into()
               .map_err(|e| anyhow::anyhow!("Invalid TXT label: {e}"))?,
               json_str.as_str().try_into()
               .map_err(|e| anyhow::anyhow!("Invalid TXT value: {e}"))?,
               ttl)
           .sign(keypair)
           .map_err(|e| anyhow::anyhow!("SignedPacket build failed: {e}"))
   }
   ```

   ```rust
   use pkarr::{Client, PublicKey};
   use anyhow::Result;
   use serde::Deserialize;

   pub struct CoordinatorInfo {
       pub coordinator_url: String,
   }

   /// Resolve a coordinator URL from a PKARR public key string (z32 format).
   ///
   /// Uses resolve_most_recent() to force a fresh DHT query (avoids stale cache).
   /// Latency: 500ms–2s typical. Acceptable for one-time coordinator discovery.
   ///
   /// The coordinator publishes a JSON blob under label "_blindjoin". Example:
   ///   {"type":"blindjoin-coordinator","onion":"127.0.0.1:8080","status":"idle",...}
   /// In Phase 4 the "onion" field holds a clearnet address. Phase 5 will contain .onion.
   pub async fn discover_coordinator(pkarr_pubkey: &str) -> Result<CoordinatorInfo> {
       let public_key: PublicKey = pkarr_pubkey.try_into()
           .map_err(|e| anyhow::anyhow!("Invalid PKARR public key '{pkarr_pubkey}': {e}"))?;

       let client = Client::builder().build()?;

       let packet = client.resolve_most_recent(&public_key).await
           .ok_or_else(|| anyhow::anyhow!(
               "Coordinator not found in DHT for key '{pkarr_pubkey}'"
           ))?;

       // Parse the _blindjoin TXT record JSON to extract the "onion" field.
       // pkarr TXT rdata is DNS wire format; collect all bytes from resource records.
       let coordinator_addr = packet
           .resource_records("_blindjoin")
           .find_map(|rr| parse_onion_from_rr(&rr))
           .ok_or_else(|| anyhow::anyhow!(
               "No 'onion' field found in PKARR record for key '{pkarr_pubkey}'"
           ))?;

       // Phase 4: coordinator_addr is "host:port" clearnet. Phase 5: will be .onion.
       // Build http:// URL for the existing CoordinatorClient.
       let coordinator_url = if coordinator_addr.starts_with("http") {
           coordinator_addr
       } else {
           format!("http://{coordinator_addr}")
       };

       tracing::info!(
           coordinator_url = %coordinator_url,
           "Resolved coordinator via PKARR DHT"
       );

       Ok(CoordinatorInfo { coordinator_url })
   }

   /// Extract the "onion" field value from a DNS TXT resource record's rdata.
   ///
   /// pkarr TXT rdata is raw DNS wire bytes. Each TXT string is length-prefixed.
   /// We concatenate all strings, interpret as UTF-8, then parse JSON.
   fn parse_onion_from_rr(rr: &pkarr::dns::ResourceRecord) -> Option<String> {
       use pkarr::dns::rdata::RData;
       let txt = match &rr.rdata {
           RData::TXT(txt) => txt,
           _ => return None,
       };
       // TXT strings are Vec<&[u8]>-like; join all character strings
       let bytes: Vec<u8> = txt.strings().flat_map(|s| s.iter().copied()).collect();
       let s = std::str::from_utf8(&bytes).ok()?;
       // Parse as JSON and extract "onion"
       #[derive(Deserialize)]
       struct Partial { onion: Option<String> }
       serde_json::from_str::<Partial>(s).ok()?.onion
   }

   #[cfg(test)]
   mod tests {
       use super::*;

       #[tokio::test]
       async fn test_invalid_pubkey_returns_error() {
           let err = discover_coordinator("not-a-valid-pkarr-key").await.unwrap_err();
           assert!(err.to_string().contains("Invalid PKARR public key"),
               "got: {err}");
       }

       // Note: test for "not found in DHT" requires a real DHT query with an unknown key.
       // This is a live network test; run manually with:
       //   cargo test --lib -p client -- discover::tests --nocapture
       // For CI, the invalid-key test is sufficient to verify error path wiring.
   }
   ```

## Trust Boundaries

|----------|-------------|
| DHT → coordinator | Unsigned DHT responses could contain malicious coordinator addresses |
| DHT → client | Client resolves coordinator URL from untrusted DHT record |
| key file → process | PKARR private key file read at startup; file permissions matter |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | Spoofing | pkarr record resolution (client) | mitigate | SignedPacket is verified by pkarr crate against the Ed25519 public key the user supplied — forged records from DHT relays are rejected. Client only resolves against user-provided key. |
| T-04-02 | Tampering | coordinator_pkarr.key file | mitigate | File should be chmod 600. Document in blindjoin.toml.example: `pkarr_key_file = "coordinator_pkarr.key"`. Docker volume mount ensures it persists across container restarts. |
| T-04-03 | Information disclosure | PKARR TXT record | accept | Record is intentionally public (coordinator address, denomination, status). No PII — no participant data, no keys, no IP of any participant. PRIV-02 satisfied. |
| T-04-04 | Denial of service | DHT publish rate | accept | 5-minute heartbeat interval avoids relay rate limits. Mitigation: log PKARR publish failures as warnings, do not crash coordinator on DHT unavailability. |
| T-04-05 | Elevation of privilege | coordinator_public_addr config field | mitigate | coordinator_public_addr is operator-supplied via config file, not derived from user input. No injection surface. Validate it is non-empty at startup (warn if default). |
| T-04-06 | Spoofing | client resolves stale cached DHT record | mitigate | Use `resolve_most_recent()` (not `resolve()`) for coordinator discovery — forces synchronous DHT query, bypasses stale cache. Per RESEARCH.md Pitfall 5. |
