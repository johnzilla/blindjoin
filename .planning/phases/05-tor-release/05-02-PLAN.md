```rust
//! Tor circuit isolation for blindjoin client.
//!
//! CLI-05: input registration and output registration use distinct Tor circuits.
//! The isolation is achieved via `TorClient::isolated_client()` which guarantees
//! that the two handles share no guard nodes or circuits.
//!
//! Source: docs.rs/arti-client/latest/arti_client/struct.TorClient.html [VERIFIED]

use anyhow::Context;
use arti_client::{TorClient, TorClientConfig};

/// Holds the bootstrapped TorClient with two isolated handles:
/// - `alice`: used for input registration (links operator to UTXO — must be isolated)
/// - `bob`: used for output registration (links to fresh output — must be isolated)
///
/// Both handles are obtained via `isolated_client()` so they provably share no circuits.
pub struct TorHandle {
    /// Isolated client for input registration phase (Alice role)
    alice: TorClient,
    /// Isolated client for output registration phase (Bob role)
    bob: TorClient,
    /// The coordinator's .onion URL (scheme + host, no trailing slash)
    coordinator_url: String,
}

impl TorHandle {
    /// Bootstrap Tor and create two isolated client handles.
    ///
    /// `coordinator_url` must be the full .onion URL, e.g. `http://xyz.onion`.
    pub async fn new(coordinator_url: String) -> anyhow::Result<Self> {
        tracing::info!("Bootstrapping Tor — this may take 10-30 seconds");
        let base = TorClient::create_bootstrapped(TorClientConfig::default())
            .await
            .context("Tor bootstrap failed — check network connectivity")?;

        // isolated_client() guarantees distinct guard nodes + circuits (CLI-05)
        let alice = base.isolated_client();
        let bob = base.isolated_client();

        tracing::info!("Tor ready — two isolated circuits allocated");
        Ok(Self { alice, bob, coordinator_url })
    }

    /// Returns the coordinator URL for the Alice (input registration) circuit.
    /// The caller uses this URL to build a reqwest::Client proxied through arti SOCKS5.
    pub fn alice_url(&self) -> &str {
        &self.coordinator_url
    }

    /// Returns the coordinator URL for the Bob (output registration) circuit.
    pub fn bob_url(&self) -> &str {
        &self.coordinator_url
    }

    /// Launch a SOCKS5 proxy listener on a local port for the alice circuit.
    /// Returns the localhost SOCKS5 proxy URL (e.g. "socks5h://127.0.0.1:PORT").
    ///
    /// The caller configures reqwest::Client::builder().proxy(proxy_url) with this.
    pub async fn alice_proxy_url(&self) -> anyhow::Result<String> {
        let port = self.launch_socks5(&self.alice).await?;
        Ok(format!("socks5h://127.0.0.1:{port}"))
    }

    /// Launch a SOCKS5 proxy listener on a local port for the bob circuit.
    pub async fn bob_proxy_url(&self) -> anyhow::Result<String> {
        let port = self.launch_socks5(&self.bob).await?;
        Ok(format!("socks5h://127.0.0.1:{port}"))
    }

    async fn launch_socks5(&self, client: &TorClient) -> anyhow::Result<u16> {
        // Bind SOCKS5 listener on a random ephemeral port (OS assigns port 0)
        let listener = client
            .launch_socks5_listener("127.0.0.1:0".parse()?)
            .await
            .context("Failed to launch arti SOCKS5 listener")?;
        let addr = listener.bind_addr();
        // Spawn the listener to serve in background
        tokio::spawn(async move {
            if let Err(e) = listener.run().await {
                tracing::warn!(error = %e, "SOCKS5 listener exited");
            }
        });
        Ok(addr.port())
    }
}

/// Convenience: bootstrap Tor and return a TorHandle.
pub async fn init_tor(coordinator_url: String) -> anyhow::Result<TorHandle> {
    TorHandle::new(coordinator_url).await
}
```

NOTE: The `launch_socks5_listener` API name may differ in arti-client 0.41. Check docs.rs for the exact method name — alternatives are `launch_socks_proxy`, `create_socks_proxy`. If the SOCKS5 approach is not available in 0.41, fall back to the `IsolationToken` + `StreamPrefs` approach (Pattern 3B in RESEARCH.md) and document the approach taken in a comment. The key invariant is: alice's HTTP requests and bob's HTTP requests MUST use circuits that cannot be linked by a network observer.

**Step 1 — client/src/http.rs**: Add `new_tor` constructor and per-phase client routing.

The existing `CoordinatorClient` uses a single `client: Client` and `base_url: String`. Extend it to optionally carry a second reqwest client for the Bob (output reg) phase:

```rust
use reqwest::{Client, Proxy};
use shared::protocol::*;
use anyhow::Result;

pub struct CoordinatorClient {
    /// Alice client — used for: poll, input reg, signing
    alice_client: Client,
    /// Bob client — used for: output registration ONLY (different Tor circuit)
    /// None in clearnet mode (alice_client handles all requests)
    bob_client: Option<Client>,
    base_url: String,
}

impl CoordinatorClient {
    /// Clearnet constructor (no change in behavior from Phase 4)
    pub fn new(base_url: String) -> Self {
        Self {
            alice_client: Client::new(),
            bob_client: None,
            base_url,
        }
    }

    /// Tor constructor — alice and bob use separate reqwest clients routed through
    /// isolated arti SOCKS5 proxies (CLI-05).
    pub fn new_tor(base_url: String, alice_proxy: String, bob_proxy: String) -> Result<Self> {
        let alice_client = Client::builder()
            .proxy(Proxy::all(&alice_proxy)?)
            .build()?;
        let bob_client = Client::builder()
            .proxy(Proxy::all(&bob_proxy)?)
            .build()?;
        Ok(Self {
            alice_client,
            bob_client: Some(bob_client),
            base_url,
        })
    }

    fn bob(&self) -> &Client {
        self.bob_client.as_ref().unwrap_or(&self.alice_client)
    }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client → Tor network (alice circuit) | Input registration flows through circuit 1 — links client to UTXO |
| Client → Tor network (bob circuit) | Output registration flows through circuit 2 — links client to fresh output |
| arti SOCKS5 localhost proxy | Both proxies bind to 127.0.0.1 only; not accessible from network |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-06 | Information Disclosure | Circuit sharing between alice and bob | mitigate | `isolated_client()` guarantees distinct circuits; `IsolationToken::no_isolation()` is never used (documented anti-pattern in RESEARCH.md) |
| T-05-07 | Spoofing | Coordinator URL in --tor mode | accept | Client validates coordinator response via existing protocol checks (round_id, blind sig verification) — Tor provides transport anonymity, not coordinator authentication |
| T-05-08 | Denial of Service | SOCKS5 proxy port binding | mitigate | Proxies bind 127.0.0.1:0 (ephemeral port, loopback only) — not network-accessible; each SOCKS5 listener is circuit-isolated so one blocked circuit doesn't affect the other |
| T-05-09 | Information Disclosure | Tor bootstrap leaks metadata to Tor directory | accept | Directory lookups are inherent to Tor; arti uses vanguards and guard selection to minimize correlation; no user PII in bootstrap packets |
| T-05-10 | Tampering | reqwest Proxy override per-client | mitigate | `Proxy::all()` routes ALL traffic for that reqwest::Client through the SOCKS5 proxy — no clearnet fallback possible for that client instance |
</threat_model>
