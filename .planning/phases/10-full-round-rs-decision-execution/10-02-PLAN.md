let deadline = tokio::time::Instant::now() + Duration::from_secs(10);
loop {
    let txids = tokio::task::spawn_blocking({
        let rpc_url = rpc_url.clone();
        let rpc_user = rpc_user.clone();
        let rpc_pass = rpc_pass.clone();
        move || {
            use corepc_node::client::client_sync::Auth;
            let auth = Auth::UserPass(rpc_user, rpc_pass);
            let client = corepc_node::Client::new_with_auth(&rpc_url, auth)
                .expect("create rpc client for mempool poll");
            client.get_raw_mempool().expect("get_raw_mempool").0
        }
    }).await.expect("mempool poll spawn_blocking");

    if !txids.is_empty() { break; }
    if tokio::time::Instant::now() >= deadline {
        panic!("mempool never received broadcast tx within 10s");
    }
    tokio::time::sleep(Duration::from_millis(100)).await;
}
