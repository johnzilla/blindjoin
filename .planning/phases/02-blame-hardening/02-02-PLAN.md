<objective>
Wire blame detection into the round lifecycle and add append-only ban file persistence.

Purpose: Without this plan, blame detection (plan 01) exists as pure logic but nothing calls it. This plan connects the signing timeout and output registration timeout to that logic, implements the round restart (Blame→Idle with survivor list), and adds disk persistence so bans survive coordinator restarts.

Output:
- Signing timeout fires → detect_non_signers → ban them → append to ban file → Blame→Idle
- Output reg timeout fires → has_missing_outputs → Blame→Idle (no individual banning — outputs are anonymous)
- Startup: read ban_list.jsonl, filter expired, load into BanList
- Blame round cap: max 2 consecutive blame rounds per original round; after that, full abort
</objective>

```rust
use serde::{Deserialize, Serialize};
use sha2::{Sha256, Digest};

/// On-disk record format (one JSON line per ban event).
/// utxo_hash = hex(SHA-256(utxo_str.as_bytes())) — raw outpoints are NOT persisted.
#[derive(Debug, Serialize, Deserialize)]
pub struct BanRecord {
    pub utxo_hash: String,
    pub banned_at: u64,
    pub expires_at: u64,
}

/// Hash a utxo_str for on-disk storage (privacy: raw outpoints not written to disk).
pub fn hash_utxo_str(utxo_str: &str) -> String {
    let digest = Sha256::digest(utxo_str.as_bytes());
    hex::encode(digest)
}

/// Append a single ban record to the ban file (append-only, newline-delimited JSON).
/// Creates the file if it does not exist.
/// BLAME-05.
pub fn append_ban_entry(path: &str, utxo_str: &str, entry: &BanEntry) -> std::io::Result<()> {
    use std::io::Write;
    let record = BanRecord {
        utxo_hash: hash_utxo_str(utxo_str),
        banned_at: entry.banned_at,
        expires_at: entry.expires_at,
    };
    let line = serde_json::to_string(&record)
        .map_err(|e| std::io::Error::new(std::io::ErrorKind::Other, e))?;
    let mut file = std::fs::OpenOptions::new()
        .create(true)
        .append(true)
        .open(path)?;
    writeln!(file, "{}", line)
}

/// Load all unexpired ban entries from the ban file.
/// Returns Ok(vec![]) if the file does not exist (first startup).
/// Skips unparseable lines with a tracing::warn!.
/// BLAME-05, BLAME-06.
pub fn load_unexpired_entries(path: &str, now_secs: u64) -> std::io::Result<Vec<(String, BanEntry)>> {
    use std::io::{BufRead, BufReader};
    let file = match std::fs::File::open(path) {
        Ok(f) => f,
        Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(vec![]),
        Err(e) => return Err(e),
    };
    let reader = BufReader::new(file);
    let mut result = Vec::new();
    for line in reader.lines() {
        let line = line?;
        let line = line.trim();
        if line.is_empty() { continue; }
        match serde_json::from_str::<BanRecord>(line) {
            Ok(record) if now_secs < record.expires_at => {
                result.push((record.utxo_hash.clone(), BanEntry {
                    banned_at: record.banned_at,
                    expires_at: record.expires_at,
                }));
            }
            Ok(_) => {} // expired — skip silently
            Err(_) => {
                tracing::warn!("ban_file: skipping unparseable line");
            }
        }
    }
    Ok(result)
}
```

2. **on_signing_timeout function** — Add to `coordinator/src/round/signing.rs`:

```rust
/// Called when the signing timeout fires. Detects non-signers, bans them,
/// appends to ban file, and transitions Signing→Blame→Idle.
///
/// `ban_list` is the Arc<RwLock<BanList>> from AppState.
/// `blame_round_count` is incremented here; if it reaches 2, full abort (no restart).
/// `ban_file_path` is from config.
///
/// BLAME-01, BLAME-04.
pub fn on_signing_timeout(
    state: &mut RoundState,
    ban_list: &mut BanList,
    ban_file_path: &str,
    ban_duration_secs: u64,
    blame_round_count: u32,
) -> BlameOutcome {
    use crate::round::blame::{detect_non_signers, append_ban_entry, now_unix_secs, BanEntry};
    use crate::round::state::Phase;
    use std::time::Duration;

    let non_signers = if let Some(inner) = &state.inner {
        detect_non_signers(&inner.registered_inputs, &inner.partial_sigs)
    } else {
        vec![]
    };

    let now = now_unix_secs();
    let ban_duration = Duration::from_secs(ban_duration_secs);

    for utxo_str in &non_signers {
        ban_list.ban(utxo_str, now, ban_duration);
        let entry = BanEntry { banned_at: now, expires_at: now + ban_duration_secs };
        if let Err(e) = append_ban_entry(ban_file_path, utxo_str, &entry) {
            tracing::warn!(ban_file = ban_file_path, "Failed to append ban entry: {e}");
        }
    }

    // Transition to Blame then Idle (zeroes round state)
    let _ = state.transition_to(Phase::Blame);
    let _ = state.transition_to(Phase::Idle);

    // Per Pitfall 3 in PITFALLS.md: cap consecutive blame rounds at 2
    if blame_round_count >= 2 {
        tracing::warn!("blame round cap reached — full round abort");
        BlameOutcome::FullAbort
    } else if non_signers.is_empty() {
        BlameOutcome::FullAbort // signing timeout fired but nobody was missing
    } else {
        BlameOutcome::RestartWithout { banned_utxos: non_signers }
    }
}
```

```rust
/// Called when the output registration timeout fires.
/// If all participants registered outputs, advance to Signing.
/// If any outputs are missing, blame and restart (missing outputs are anonymous — no individual banning).
/// BLAME-02, BLAME-04.
pub fn on_output_reg_timeout(state: &mut RoundState) -> OutputRegOutcome {
    use crate::round::blame::has_missing_outputs;
    use crate::round::state::Phase;

    let (input_count, output_count) = if let Some(inner) = &state.inner {
        (inner.registered_inputs.len(), inner.registered_outputs.len())
    } else {
        (0, 0)
    };

    if has_missing_outputs(input_count, output_count) {
        // Cannot ban individually (outputs are anonymous). Blame→Idle, restart.
        let _ = state.transition_to(Phase::Blame);
        let _ = state.transition_to(Phase::Idle);
        return OutputRegOutcome::BlameRestart;
    }

    // All outputs present — advance to Signing
    let _ = state.transition_to(Phase::Signing);
    OutputRegOutcome::AdvanceToSigning
}

pub enum OutputRegOutcome {
    AdvanceToSigning,
    BlameRestart,
}
```

```rust
// In the tokio::spawn for signing phase timer:
let ban_list_clone = Arc::clone(&app_state.ban_list);
let blame_count_clone = Arc::clone(&app_state.blame_round_count);
let ban_file = config.coordinator.ban_file_path.clone();

tokio::spawn(run_phase_timer(
    Arc::clone(&round_state),
    Phase::Signing,
    Duration::from_secs(config.coordinator.round_timeout_signing_secs),
    move |state| {
        let mut ban_list = ban_list_clone.blocking_lock(); // sync callback context
        let count = blame_count_clone.load(std::sync::atomic::Ordering::Relaxed);
        let outcome = on_signing_timeout(state, &mut ban_list, &ban_file, ban_duration, count);
        match outcome {
            BlameOutcome::FullAbort => {
                blame_count_clone.store(0, std::sync::atomic::Ordering::Relaxed);
            }
            BlameOutcome::RestartWithout { .. } => {
                blame_count_clone.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
                // Restart: transition back to InputReg for next round
                // (RoundState is now Idle after on_signing_timeout; next poll will see Idle)
            }
        }
    },
));
```

**Simpler approach** (recommended to avoid lock ordering issues): After `run_phase_timer` returns (it's async and awaited in a spawned task), check state phase and run blame detection outside the timer closure. The pattern:

```rust

    // Wait for signing phase to time out
    tokio::time::sleep(signing_timeout).await;

        return; // already advanced
```
