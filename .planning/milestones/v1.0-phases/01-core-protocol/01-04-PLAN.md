use std::sync::Arc;
use tokio::sync::RwLock;
use tracing::{error, info};

mod api;
mod bitcoin;
mod blind;
mod config;
mod round;

use config::CoordinatorConfig;
use bitcoin::rpc::BitcoinRpc;
use round::state::RoundState;

#[tokio::main]
async fn main() -> anyhow::Result<()> {
    // Initialize structured logging — never log PII (PRIV-02)
    tracing_subscriber::fmt()
        .with_env_filter(
            tracing_subscriber::EnvFilter::from_default_env()
                .add_directive("coordinator=info".parse().unwrap()),
        )
        .json()  // structured JSON logs for production
        .init();

    // Load configuration (D-11)
    let cfg = CoordinatorConfig::load().unwrap_or_else(|e| {
        error!(error = %e, "Config load failed — using defaults");
        CoordinatorConfig::with_defaults()
    });

    info!(
        network = %cfg.network.bitcoin_network,
        denomination_sats = cfg.coordinator.denomination_sats,
        min_participants = cfg.coordinator.min_participants,
        "Coordinator starting"
    );

    // Fail-fast startup health checks (D-12)
    let rpc = BitcoinRpc::new(
        cfg.network.bitcoin_rpc_url.clone(),
        cfg.network.bitcoin_rpc_user.clone(),
        cfg.network.bitcoin_rpc_pass.clone(),
    );
    startup_health_check(&rpc, &cfg.network.bitcoin_network).await?;

    // Initialize shared round state
    let round_state: Arc<RwLock<RoundState>> = Arc::new(RwLock::new(RoundState::new_idle()));

    // Build and run the axum server
    let app = api::build_router(round_state.clone(), Arc::new(rpc), Arc::new(cfg.clone()));
    let listener = tokio::net::TcpListener::bind(&cfg.coordinator.listen_addr).await?;
    info!(addr = %cfg.coordinator.listen_addr, "Listening");
    axum::serve(listener, app).await?;

    Ok(())
}

async fn startup_health_check(rpc: &BitcoinRpc, expected_network: &str) -> anyhow::Result<()> {
    // 1. Verify bitcoind reachable
    let block_count = rpc.getblockcount().await.map_err(|e| {
        anyhow::anyhow!("Bitcoin Core unreachable at startup: {e}. Is bitcoind running?")
    })?;
    info!(block_count, "Bitcoin Core reachable");

    // 2. Verify network matches config
    // getblockchaininfo returns chain name ("signet", "test", "main")
    // For Phase 1, we trust the configured RPC URL targets the right network.
    // A more robust check would call getblockchaininfo — implement if needed.

    // 3. Block count > 0 means not in a trivially bad state (not checking IBD status
    // here since signet IBD is fast — add getblockchaininfo check if needed in Phase 2)
    if block_count == 0 {
        anyhow::bail!("Bitcoin Core reports 0 blocks — may be misconfigured or not synced");
    }

    Ok(())
}

use std::sync::Arc;
use tokio::sync::RwLock;
use axum::{Router, routing::{get, post}};
use tower_http::limit::RequestBodyLimitLayer;
use crate::round::state::RoundState;
use crate::bitcoin::rpc::BitcoinRpc;
use crate::config::CoordinatorConfig;

pub mod handlers;
pub mod middleware;

pub fn build_router(
    round: Arc<RwLock<RoundState>>,
    rpc: Arc<BitcoinRpc>,
    config: Arc<CoordinatorConfig>,
) -> Router {
    Router::new()
        .route("/info", get(handlers::get_info))
        .route("/round/input", post(handlers::post_input))
        .route("/round/output", post(handlers::post_output))
        .route("/round/sign", post(handlers::post_sign))
        .route("/round/tx", get(handlers::get_tx))
        .layer(RequestBodyLimitLayer::new(64 * 1024)) // 64KB max request body
        .with_state(AppState { round, rpc, config })
}

#[derive(Clone)]
pub struct AppState {
    pub round: Arc<RwLock<RoundState>>,
    pub rpc: Arc<BitcoinRpc>,
    pub config: Arc<CoordinatorConfig>,
}

pub async fn get_info(State(state): State<AppState>) -> Json<InfoResponse> {
    let guard = state.round.read().await;
    // Populate rsa_pubkey_der_b64 from RoundState.rsa_pubkey_der (Option<Vec<u8>>).
    // RoundState must be extended with `pub rsa_pubkey_der: Option<Vec<u8>>`
    // populated on InputReg transition alongside rsa_pubkey_hash.
    // Both are derived from RsaBlindSigner and cleared on Idle transition.
    let rsa_pubkey_der_b64 = guard.rsa_pubkey_der
        .as_ref()
        .map(|der| base64::engine::general_purpose::STANDARD.encode(der));
    Json(InfoResponse {
        version: env!("CARGO_PKG_VERSION").to_string(),
        network: state.config.network.bitcoin_network.clone(),
        denomination_sats: state.config.coordinator.denomination_sats,
        min_participants: state.config.coordinator.min_participants,
        max_participants: state.config.coordinator.max_participants,
        round_state: guard.phase.as_str().to_string(),
        participants_registered: guard.participant_count,
        rsa_pubkey_hash: guard.rsa_pubkey_hash.map(hex::encode),
        rsa_pubkey_der_b64,  // None when Idle; clients use this to blind tokens (D-02)
        round_id: Some(guard.round_id),
    })
}

**POST /round/input:** In INPUT_REG phase:
- Parse InputRegRequest
- Parse utxo_outpoint string "txid:vout" into OutPoint
- Decode ownership_proof via OwnershipProof::from_json_hex_str() (canonical JSON-array-of-hex-strings format, not raw base64)
- Decode base64 blinded_token → BlindedMessage
- Call validate_utxo (checks existence, value, double-reg, BIP-322) — return appropriate error
- Write lock: call blind_sign on the round's RsaBlindSigner
- Generate session token via generate_session_token
- Add to registered_inputs
- Increment participant_count
- If participant_count >= max_participants, advance phase to OutputReg (early advance)
- Return InputRegResponse { blind_signature (base64), round_id, session_token (base64) }

**POST /round/output:** In OUTPUT_REG phase:
- Parse OutputRegRequest
- Decode base64 signature → unblinded RSA signature bytes
- Decode base64 unblinded_token → 32-byte message hash
- Verify RSA signature on unblinded_token using round's public key
- Check unblinded_token not in redeemed_tokens set (replay prevention, TEST-04)
- Verify token matches the domain-separated format (optional — the RSA sig already binds it)
- Parse output_address → ScriptBuf
- Verify amount_sats == denomination (D-16, PROTO-03)
- Write lock: add to registered_outputs, add token hash to redeemed_tokens
- If all participants have registered outputs, advance to Signing phase
- Return OutputRegResponse { accepted: true, round_id }

**GET /round/tx:** In SIGNING phase:
- Read lock: call build_coinjoin_psbt with registered_inputs + registered_outputs
- Return RoundTxResponse { round_id, psbt (base64), fee_total_sats, fee_per_participant_sats }

**POST /round/sign:** In SIGNING phase:
- Parse SignRequest { round_id, utxo_outpoint, partial_signature, session_token }
- Write lock: verify session_token via verify_session_token
- Verify utxo_outpoint is in registered_inputs (not from a different participant)
- Store partial_signature in partial_sigs keyed by utxo_outpoint
- If partial_sigs.len() == participant_count: advance to Broadcast phase, assemble + broadcast TX
- On broadcast success: log txid (ALLOWED — it's public), transition to Idle, zero inner state
- Return 200 OK

```rust
/// Pure output registration logic — callable from tests without axum/state machinery.
///
/// Arguments:
///   pk             — round RSA public key (for signature verification)
///   redeemed       — mutable token replay set (appended if token accepted)
///   token_msg      — the 32-byte unblinded token message M
///   sig_bytes      — the unblinded RSA signature bytes
///   denomination   — configured denomination_sats
///   amount_sats    — amount_sats from the request
pub fn register_output_logic(
    pk: &blind_rsa_signatures::PublicKey,
    redeemed: &mut Vec<[u8; 32]>,
    token_msg: &[u8; 32],
    sig_bytes: &[u8],
    denomination: u64,
    amount_sats: u64,
) -> Result<(), shared::errors::ApiError> {
    use shared::errors::{ApiError, ErrorCode};
    use blind_rsa_signatures::{Signature, Options};

    // 1. Check denomination match (PROTO-03)
    if amount_sats != denomination {
        return Err(ApiError { code: ErrorCode::WrongDenomination,
            message: format!("Expected {denomination} sats, got {amount_sats}"),
            round_id: None });
    }

    // 2. Replay check (PROTO-04)
    if redeemed.contains(token_msg) {
        return Err(ApiError { code: ErrorCode::TokenAlreadyUsed,
            message: "Token already redeemed".into(),
            round_id: None });
    }

    // 3. RSA signature verification
    let options = Options::default();
    let sig = Signature::from_bytes(sig_bytes)
        .map_err(|_| ApiError { code: ErrorCode::InvalidToken,
            message: "Invalid token signature format".into(),
            round_id: None })?;
    pk.verify(&sig, None, token_msg, &options)
        .map_err(|_| ApiError { code: ErrorCode::InvalidToken,
            message: "Token signature verification failed".into(),
            round_id: None })?;

    // 4. Mark token as redeemed
    redeemed.push(*token_msg);
    Ok(())
}
```
