Implementation (based on verified arti-axum source + docs.rs/arti-client 0.41 API):
```rust
use arti_client::{TorClient, TorClientConfig};
use arti_client::config::onion_service::OnionServiceConfigBuilder;
use tor_hsservice::{handle_rend_requests, IncomingStreamRequest};
use hyper_util::rt::TokioIo;
use hyper::server::conn::http1;
use futures_util::StreamExt;
use anyhow::Context;

pub async fn serve_onion_service(
    app: axum::Router,
    addr_tx: tokio::sync::oneshot::Sender<String>,
) -> anyhow::Result<()> {
    let tor_client = TorClient::create_bootstrapped(TorClientConfig::default())
        .await
        .context("Failed to bootstrap Tor — check network connectivity")?;

    let hs_config = OnionServiceConfigBuilder::default()
        .nickname("blindjoin".to_owned().try_into()
            .context("Invalid HS nickname")?)
        .build()
        .context("Failed to build onion service config")?;

    let (onion_service, rend_requests) = tor_client
        .launch_onion_service(hs_config)?
        .expect("onion-service-service feature must be enabled in Cargo.toml");

    let onion_addr = onion_service
        .onion_name()
        .expect("Onion address available immediately after launch")
        .to_string();

    tracing::info!(
        addr = %onion_addr,
        "Tor hidden service launched — descriptor propagation may take 30-60s"
    );

    // Send onion address to main.rs for PKARR publishing
    let _ = addr_tx.send(onion_addr);

    let stream_requests = handle_rend_requests(rend_requests);
    tokio::pin!(stream_requests);

    while let Some(stream_req) = stream_requests.next().await {
        let IncomingStreamRequest::Begin(_) = stream_req.request() else {
            continue;
        };
        let data_stream = match stream_req.accept(
            tor_hsservice::ConnectedFlags::new_empty()
        ).await {
            Ok(ds) => ds,
            Err(e) => {
                tracing::warn!(error = %e, "Failed to accept HS stream");
                continue;
            }
        };
        let io = TokioIo::new(data_stream);
        let app_clone = app.clone();
        tokio::spawn(async move {
            if let Err(e) = http1::Builder::new()
                .serve_connection(io, app_clone)
                .await
            {
                tracing::debug!(error = %e, "HS connection closed");
            }
        });
    }
    Ok(())
}
```

NOTE: The exact API for `stream_req.accept(...)` argument may differ slightly between arti 0.41 versions. If the `ConnectedFlags` type doesn't exist, try `tor_hsservice::Connected::new_empty()` or the unit argument `()` — check the compiler error message and adjust. Document the exact type used as a comment.

// Determine the coordinator's public address for PKARR.
// In tor_mode: wait for onion address from the hidden service task.
// In clearnet mode: use cfg.discovery.coordinator_public_addr directly.
let public_addr: String = if cfg.coordinator.tor_mode {
    // Channel: tor.rs sends the .onion address after launch_onion_service returns
    let (addr_tx, addr_rx) = oneshot::channel::<String>();

    // Build router BEFORE spawning the onion service task
    let app = api::build_router_with_ban_list(
        round_state.clone(),
        Arc::new(rpc),
        Arc::new(cfg.clone()),
        ban_list,
    );

    // Spawn the hidden service — it sends the onion address then serves forever
    tokio::spawn(async move {
        if let Err(e) = serve_onion_service(app, addr_tx).await {
            error!(error = %e, "Onion service fatal error");
            std::process::exit(1);
        }
    });

    // Wait for the onion address (arrives within ~1s of Tor bootstrap completing)
    addr_rx.await.context("Onion service task exited before sending address")?

    // Clearnet path (Phase 4 compatible — default for tests)
    let app = api::build_router_with_ban_list(

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Tor network → onion service | All inbound HTTP arrives via Tor circuits — origin IP is unlinkable by design |
| onion service → axum handlers | DataStream wrapped as hyper IO — untrusted data; axum middleware applies same validation as clearnet path |
| config file → tor_mode flag | Config is operator-controlled; setting tor_mode=false in production would expose clearnet — must be documented |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-01 | Spoofing | serve_onion_service / RendRequest accept | mitigate | Only `IncomingStreamRequest::Begin` variant is accepted; all other variants are skipped with `continue` — prevents non-HTTP stream injection |
| T-05-02 | Tampering | config tor_mode default | mitigate | Default is `false` (safe for dev); production config explicitly sets `tor_mode = true` in blindjoin.toml — documented in .env.example |
| T-05-03 | Information Disclosure | PKARR publish timing | accept | Onion address published immediately after `launch_onion_service` before descriptor propagates; first connections may fail for 30-60s — clients already have retry logic (polling design per PRIV-04) |
| T-05-04 | Denial of Service | TorClient bootstrap blocking main thread | mitigate | `create_bootstrapped` runs in `tokio::spawn` inside `serve_onion_service`; if bootstrap fails, error is logged and process exits 1 — no silent hang |
| T-05-05 | Elevation of Privilege | clearnet TCP listener left open in tor_mode | mitigate | tor_mode branch never calls `TcpListener::bind`; clearnet branch never calls `serve_onion_service` — mutually exclusive code paths enforced at compile time by structure (no shared bind call) |
</threat_model>
